   937→class FeatureAdder(BaseEstimator, TransformerMixin):
   938→    def fit(self, X, y=None): return self
   939→    def transform(self, X):
   940→        rooms, hh, pop = X[:, 3], X[:, 4], X[:, 5]
   941→        return np.c_[X, rooms / hh, pop / hh, rooms / pop]
