    91→
    92→export function RegisterForm() {
    93→  const [step, setStep] = React.useState(1);
    94→  const [category, setCategory] = React.useState<Category | null>(null);
    95→  const [project, setProject] = React.useState<Project | null>(null);
    96→  const [search, setSearch] = React.useState("");
    97→  const [form, setForm] = React.useState<FormState>(EMPTY_FORM);
    98→  const [errors, setErrors] = React.useState<Partial<Record<keyof FormState, string>>>({});
    99→  const [consentWa, setConsentWa] = React.useState(true);
   100→  const [consentEm, setConsentEm] = React.useState(true);
   101→  const [submitting, setSubmitting] = React.useState(false);
   102→  const [result, setResult] = React.useState<SubmitResult | null>(null);
   103→  const { toast } = useToast();
   104→
   105→  // Filtered projects in current category
   106→  const filteredProjects = React.useMemo(() => {
   107→    if (!category) return [];
   108→    const q = search.trim().toLowerCase();
   109→    if (!q) return category.projects;
   110→    return category.projects.filter(
   111→      (p) => p.name.toLowerCase().includes(q) || p.tags.toLowerCase().includes(q)
   112→    );
   113→  }, [category, search]);
   114→
   115→  function pickCategory(c: Category) {
   116→    setCategory(c);
   117→    setProject(null); // reset project selection when category changes
   118→    setSearch("");
   119→  }
   120→
   121→  function canNext(): boolean {
   122→    if (step === 1) return !!category;
   123→    if (step === 2) return !!project;
   124→    if (step === 3) return Object.keys(errors).length === 0 && validateStep3();
   125→    return true;
   126→  }
   127→
   128→  function validateStep3(): boolean {
   129→    const e: Partial<Record<keyof FormState, string>> = {};
   130→    if (!form.firstName.trim()) e.firstName = "First name is required";
   131→    if (!form.lastName.trim()) e.lastName = "Last name is required";
   132→    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(form.email))
   133→      e.email = "Valid email required";
   134→    if (!/^[+\d][\d\s-]{8,}$/.test(form.mobile))
   135→      e.mobile = "Valid mobile number required";
   136→    if (!form.degree) e.degree = "Select your degree";
   137→    if (!form.branch) e.branch = "Select your branch";
   138→    if (!form.semester) e.semester = "Select semester";
   139→    if (!form.plan) e.plan = "Select a plan";
   140→    if (!form.college.trim()) e.college = "College name required";
   141→    if (!form.state) e.state = "Select your state";
   142→    setErrors(e);
   143→    return Object.keys(e).length === 0;
   144→  }
   145→
   146→  function next() {
   147→    if (step === 3 && !validateStep3()) {
   148→      toast({ title: "Please fix the errors", description: "Some required fields are missing or invalid.", variant: "destructive" });
   149→      return;
   150→    }
   151→    setStep((s) => Math.min(4, s + 1));
   152→  }
   153→  function back() {
   154→    setStep((s) => Math.max(1, s - 1));
   155→  }
   156→
   157→  async function submit() {
   158→    if (!category || !project) return;
   159→    if (!consentWa && !consentEm) {
   160→      toast({
   161→        title: "Consent required",
   162→        description: "Please allow at least one contact channel (WhatsApp/phone or email) so our team can reach you.",
   163→        variant: "destructive",
   164→      });
   165→      return;
   166→    }
   167→    setSubmitting(true);
   168→    try {
   169→      const res = await fetch("/api/register", {
   170→        method: "POST",
   171→        headers: { "Content-Type": "application/json" },
   172→        body: JSON.stringify({
   173→          firstName: form.firstName,
   174→          lastName: form.lastName,
   175→          email: form.email,
   176→          mobile: form.mobile,
   177→          degree: form.degree,
   178→          branch: form.branch,
   179→          semester: form.semester,
   180→          plan: form.plan,
   181→          college: form.college,
   182→          state: form.state,
   183→          source: form.source || undefined,
   184→          notes: form.notes || undefined,
   185→          category: category.key,
   186→          categoryLabel: category.label,
   187→          project: project.name,
   188→          projectId: project.id,
   189→          consentWa,
   190→          consentEm,
   191→        }),
   192→      });
   193→      const data = await res.json();
   194→      if (!res.ok || !data.ok) {
   195→        throw new Error(data.error || "Submission failed");
   196→      }
   197→      setResult({
   198→        id: data.id,
   199→        student: data.student,
   200→        notification: data.notification,
   201→      });
   202→      toast({
   203→        title: "🎉 Enrolment Submitted!",
   204→        description: `Your registration for "${project.name}" was saved. Open WhatsApp or email to confirm with Shekhar's team.`,
   205→      });
   206→    } catch (err) {
   207→      toast({
   208→        title: "Submission failed",
   209→        description: (err as Error).message || "Please try again in a moment.",
   210→        variant: "destructive",
   211→      });
   212→    } finally {
   213→      setSubmitting(false);
   214→    }
   215→  }
   216→
   217→  function resetAll() {

   219→    setStep(1);
   220→    setCategory(null);
   221→    setProject(null);
   222→    setSearch("");
   223→    setForm(EMPTY_FORM);
   224→    setErrors({});
   225→    setConsentWa(true);
   226→    setConsentEm(true);
   227→  }
   228→
   229→  return (
   230→    <section
   231→      id="register"
   232→      className="py-16 md:py-24 px-5 md:px-[5%] max-w-7xl mx-auto scroll-mt-20"
   233→      aria-labelledby="register-h2"
   234→    >
   235→      <div className="text-center mb-10">
   236→        <div className="text-cyan-400 text-xs font-semibold tracking-widest uppercase mb-3">
   237→          Free Enrolment · 4 Steps
   238→        </div>
   239→        <h2 id="register-h2" className="font-bold text-3xl md:text-4xl">
   240→          Register for your <span className="ap-gradient-text">free internship</span>
   241→        </h2>
   242→        <p className="text-slate-400 max-w-2xl mx-auto mt-3">
   243→          Pick a track, choose a project, enter your details, and confirm. No fee, no documents, no friction — your seat is reserved the moment you submit.
   244→        </p>
   245→      </div>
   246→
   247→      <div className="bg-[#0e1b2e] border border-[rgba(6,182,212,0.22)] rounded-2xl p-5 md:p-8">
   248→        {result ? (
   249→          <SuccessPanel result={result} onReset={resetAll} />
   250→        ) : (
   251→          <>
   252→            {/* Step indicators */}
   253→            <StepIndicator current={step} />
   254→
   255→            {/* STEP 1 — Category */}

   257→              <div className="mt-8">
   258→                <StepHeader
   259→                  num={1}
   260→                  title="Select your internship category"
   261→                  desc="Pick the AI/ML domain or classic track you want to intern in. You can change this any time before submitting."
   262→                />

   269→                        key={c.key}
   270→                        onClick={() => pickCategory(c)}
   271→                        aria-pressed={active}
   272→                        className={cn(
   273→                          "ap-lift text-left p-4 rounded-xl border transition-all relative",
   274→                          active
   275→                            ? "border-cyan-400 bg-[#0F2340] ring-2 ring-cyan-500/40"
   276→                            : "border-[rgba(6,182,212,0.18)] bg-[#111f35] hover:border-cyan-500/40"
   277→                        )}

   279→                        <div className="text-2xl mb-2" aria-hidden>{c.icon}</div>
   280→                        <div className="text-sm font-semibold text-foreground leading-tight">
   281→                          {c.label}

   283→                        <div className="text-[11px] text-slate-500 mt-1">{c.sub}</div>
   284→                        {active && (
   285→                          <span className="absolute top-3 right-3 size-5 rounded-full ap-gradient-cyan grid place-items-center text-navy">
   286→                            <Check className="size-3.5" />

   289→                      </button>
   290→                    );
   291→                  })}
   292→                </div>
   293→                <WizardNav

   297→                />

   300→

   326→                  {filteredProjects.map((p) => {
   327→                    const active = project?.id === p.id;
   328→                    return (
   329→                      <button
   330→                        type="button"
   331→                        key={p.id}
   332→                        onClick={() => setProject(p)}
   333→                        aria-pressed={active}
   334→                        className={cn(
   335→                          "text-left p-3 rounded-lg border transition-all flex items-start gap-3",
   336→                          active
   337→                            ? "border-cyan-400 bg-[#0F2340] ring-2 ring-cyan-500/40"
   338→                            : "border-[rgba(6,182,212,0.18)] bg-[#111f35] hover:border-cyan-500/40"
   339→                        )}
   340→                      >
   341→                        <span className="size-6 shrink-0 rounded-md bg-[#0a1422] border border-[rgba(6,182,212,0.18)] grid place-items-center text-[10px] font-bold text-slate-400 mt-0.5">
   342→                          {p.id}

   344→                        <span className="flex-1">

   346→                            {p.name}

   349→                            {p.tags}

   355→                          )}
   356→                        </span>

   358→                    );

   360→                </div>

   379→                <StepHeader

   381→                  title="Your details"
   382→                  desc="We need your basic info to register you on the program and issue your certificate. We never sell your data — see our privacy notes below."
   383→                />
   384→                <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mt-6">

   386→                    <Input
   387→                      value={form.firstName}
   388→                      onChange={(e) => setForm({ ...form, firstName: e.target.value })}
   389→                      placeholder="e.g. Aditya"

   391→                    />

   394→                    <Input
   395→                      value={form.lastName}

   398→                      aria-invalid={!!errors.lastName}
   399→                    />
   400→                  </Field>
   401→                  <Field label="Email" error={errors.email} required>
   402→                    <Input

   409→                  </Field>
