   143→function useStudents() {
   144→  const [students, setStudents] = useState<Student[]>([]);
   145→  const [loading, setLoading] = useState(true);
   146→  const load = useCallback(async () => {
   147→    setLoading(true);
   148→    try {
   149→      const res = await fetch("/api/admin/registrations?since=all&limit=500", { cache: "no-store" });
   150→      const data = await res.json();
   151→      if (data.ok) setStudents(data.recent || []);
   152→    } catch { /* ignore */ }
   153→    finally { setLoading(false); }
   154→  }, []);
   155→  useEffect(() => { load(); }, [load]);
   156→  return { students, loading, load };
   157→}

   215→// ── Students tab (filters + actions) ──
   216→function StudentsTab() {
   217→  const { students, loading, load } = useStudents();
   218→  const { toast } = useToast();
   219→  const [busy, setBusy] = useState<string | null>(null);
   220→  const [q, setQ] = useState("");
   221→  const [filterStatus, setFilterStatus] = useState("all");
   222→  const [filterCollege, setFilterCollege] = useState("all");
   223→
   224→  const colleges = Array.from(new Set(students.map(s => s.college).filter(Boolean))).sort();
   225→  const filtered = students.filter(s => {
   226→    if (q && !(`${s.firstName} ${s.lastName} ${s.email} ${s.project}`.toLowerCase().includes(q.toLowerCase()))) return false;
   227→    if (filterStatus !== "all") {
   228→      if (filterStatus === "pending" && s.paymentStatus === "confirmed") return false;
   229→      if (filterStatus === "active" && s.internshipStatus !== "in_progress") return false;
   230→      if (filterStatus === "completed" && s.internshipStatus !== "completed") return false;
   231→    }
   232→    if (filterCollege !== "all" && s.college !== filterCollege) return false;
   233→    return true;
   234→  });
   235→
   236→  async function start(id: string, name: string) {
   237→    setBusy(id); try {
   238→      const res = await fetch("/api/admin/student/start", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ studentId: id }) });
   239→      const d = await res.json(); if (!res.ok || !d.ok) throw new Error(d.error);
   240→      toast({ title: `Approved ${name}'s payment`, description: "Stage 1 unlocked." }); await load();
   241→    } catch (e) { toast({ title: "Failed", description: (e as Error).message, variant: "destructive" }); }
   242→    finally { setBusy(null); }
   243→  }
   244→  async function advance(id: string, name: string) {
   245→    setBusy(id + "-a"); try {
   246→      const res = await fetch("/api/admin/student/advance", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ studentId: id }) });
   247→      const d = await res.json(); if (!res.ok || !d.ok) throw new Error(d.error);
   248→      const done = d.student.internshipStatus === "completed";
   249→      toast({ title: done ? `🎉 ${name} completed!` : `${name} → Stage ${d.student.currentStage}`, description: done ? "Ready for certificate." : `Stage ${d.student.currentStage}/6.` }); await load();
   250→    } catch (e) { toast({ title: "Failed", description: (e as Error).message, variant: "destructive" }); }
   251→    finally { setBusy(null); }
   252→  }
   253→
   254→  return (
   255→    <div>
   256→      <h3 className="font-bold text-lg mb-1">Student Management</h3>
   257→      <p className="text-xs text-muted-foreground mb-4">Filter, approve payments, advance stages, generate certificates.</p>
   258→      <div className="flex flex-wrap gap-2 mb-3">
   259→        <div className="relative flex-1 min-w-[180px]">
   260→          <Search className="size-4 absolute left-2.5 top-2.5 text-muted-foreground" />
   261→          <Input value={q} onChange={e => setQ(e.target.value)} placeholder="Search name / email / project…" className="bg-muted border-border pl-8 h-9 text-sm" />
   262→        </div>
   263→        <select value={filterStatus} onChange={e => setFilterStatus(e.target.value)} className="bg-muted border border-border rounded-md h-9 px-2 text-xs text-foreground">
   264→          <option value="all">All status</option><option value="pending">Pending</option><option value="active">Active</option><option value="completed">Completed</option>
   265→        </select>
   266→        <select value={filterCollege} onChange={e => setFilterCollege(e.target.value)} className="bg-muted border border-border rounded-md h-9 px-2 text-xs text-foreground max-w-[160px]">
   267→          <option value="all">All colleges</option>{colleges.map(c => <option key={c} value={c}>{c}</option>)}
   268→        </select>
   269→        <Button variant="ghost" size="sm" onClick={load} disabled={loading} className="text-foreground hover:text-cyan-500 dark:text-cyan-400 h-9"><RefreshCw className={cn("size-3.5", loading && "animate-spin")} /></Button>
   270→      </div>
   271→      {loading ? <Loading /> : filtered.length === 0 ? <Empty text="No students match the filters." /> : (
   272→        <div className="border border-border rounded-lg max-h-[60vh] overflow-y-auto">
   273→          <table className="w-full text-sm">
   274→            <thead className="sticky top-0 bg-popover z-10"><tr className="text-left text-[10px] text-muted-foreground border-b border-border">
   275→              <th className="px-3 py-2">Student</th><th className="px-3 py-2 hidden md:table-cell">Project</th><th className="px-3 py-2">Status</th><th className="px-3 py-2 text-right">Action</th>
   276→            </tr></thead>
   277→            <tbody>
   278→              {filtered.map(s => {
   279→                const done = s.internshipStatus === "completed";
   280→                const inProg = s.internshipStatus === "in_progress";
   281→                return (
   282→                  <tr key={s.id} className="border-b border-border/30 last:border-0 hover:bg-secondary/40">
   283→                    <td className="px-3 py-2"><div className="font-medium text-foreground text-xs">{s.firstName} {s.lastName}</div><div className="text-[10px] text-muted-foreground">{s.degree} · {s.college}</div></td>
   284→                    <td className="px-3 py-2 hidden md:table-cell text-xs text-foreground max-w-[180px]"><div className="truncate" title={s.project}>{s.project}</div><div className="text-[10px] text-muted-foreground">{s.categoryLabel}</div></td>
   285→                    <td className="px-3 py-2">{done ? <Badge color="emerald" icon={<CheckCircle2 className="size-2.5" />}>Done</Badge> : inProg ? <Badge color="cyan" icon={<Loader2 className="size-2.5" />}>Stage {s.currentStage}/6</Badge> : <Badge color="amber" icon={<Lock className="size-2.5" />}>Pending</Badge>}</td>
   286→                    <td className="px-3 py-2 text-right whitespace-nowrap">
   287→                      {done ? <GenCertButton studentId={s.id} name={s.firstName} small /> :
   288→                        inProg ? <Button size="sm" variant="outline" onClick={() => advance(s.id, s.firstName)} disabled={busy === s.id + "-a"} className="h-7 text-[11px] bg-secondary border-border text-foreground hover:text-cyan-500 dark:text-cyan-400">{busy === s.id + "-a" ? <Loader2 className="size-3 animate-spin" /> : <Forward className="size-3.5" />}Advance</Button> :
   289→                        <Button size="sm" onClick={() => start(s.id, s.firstName)} disabled={busy === s.id} className="h-7 text-[11px] bg-gradient-to-r from-emerald-600 to-cyan-500 text-primary-foreground hover:opacity-90">{busy === s.id ? <Loader2 className="size-3 animate-spin" /> : <Rocket className="size-3.5" />}Approve & Start</Button>}
   290→                    </td>
   291→                  </tr>
   292→                );
   293→              })}
   294→            </tbody>
   295→          </table>
   296→        </div>
   297→      )}
   298→    </div>
   299→  );
   300→}

   328→function AppGroup({ title, count, accent, students, empty, action }: { title: string; count: number; accent: "amber" | "cyan" | "emerald"; students: Student[]; empty: string; action: (s: Student) => React.ReactNode }) {
   329→  const c = { amber: "text-amber-300 border-amber-500/30", cyan: "text-cyan-500 dark:text-cyan-400 border-cyan-500/30", emerald: "text-emerald-300 border-emerald-500/30" }[accent];
   330→  return (
