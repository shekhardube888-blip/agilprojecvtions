    65→  // Listen for "view-certificate" events from nested components (GenCertButton,

    67→  useEffect(() => {
    68→    const handler = async (e: Event) => {
    69→      const number = (e as CustomEvent).detail as string;
    70→      if (!number) return;

    78→      } catch { /* ignore */ }
    79→    };

   171→function useStudents() {
   172→  const [students, setStudents] = useState<Student[]>([]);
   173→  const [loading, setLoading] = useState(true);
   174→  const load = useCallback(async () => {
   175→    setLoading(true);
   176→    try {
   177→      const res = await fetch("/api/admin/registrations?since=all&limit=500", { cache: "no-store" });
   178→      const data = await res.json();
   179→      if (data.ok) setStudents(data.recent || []);
   180→    } catch { /* ignore */ }
   181→    finally { setLoading(false); }
   182→  }, []);
   183→  useEffect(() => { load(); }, [load]);
   184→  return { students, loading, load };
   185→}

   243→// ── Students tab (filters + actions) ──
   244→function StudentsTab() {
   245→  const { students, loading, load } = useStudents();
   246→  const { toast } = useToast();
   247→  const [busy, setBusy] = useState<string | null>(null);
   248→  const [q, setQ] = useState("");
   249→  const [filterStatus, setFilterStatus] = useState("all");
   250→  const [filterCollege, setFilterCollege] = useState("all");
   251→
   252→  const colleges = Array.from(new Set(students.map(s => s.college).filter(Boolean))).sort();
   253→  const filtered = students.filter(s => {
   254→    if (q && !(`${s.firstName} ${s.lastName} ${s.email} ${s.project}`.toLowerCase().includes(q.toLowerCase()))) return false;
   255→    if (filterStatus !== "all") {
   256→      if (filterStatus === "pending" && s.paymentStatus === "confirmed") return false;
   257→      if (filterStatus === "active" && s.internshipStatus !== "in_progress") return false;
   258→      if (filterStatus === "completed" && s.internshipStatus !== "completed") return false;
   259→    }
   260→    if (filterCollege !== "all" && s.college !== filterCollege) return false;
   261→    return true;
   262→  });
   263→
   264→  async function start(id: string, name: string) {
   265→    setBusy(id); try {
   266→      const res = await fetch("/api/admin/student/start", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ studentId: id }) });
   267→      const d = await res.json(); if (!res.ok || !d.ok) throw new Error(d.error);
   268→      toast({ title: `Approved ${name}'s payment`, description: "Stage 1 unlocked." }); await load();
   269→    } catch (e) { toast({ title: "Failed", description: (e as Error).message, variant: "destructive" }); }
   270→    finally { setBusy(null); }
   271→  }
   272→  async function advance(id: string, name: string) {
   273→    setBusy(id + "-a"); try {
   274→      const res = await fetch("/api/admin/student/advance", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ studentId: id }) });
   275→      const d = await res.json(); if (!res.ok || !d.ok) throw new Error(d.error);
   276→      const done = d.student.internshipStatus === "completed";
   277→      toast({ title: done ? `🎉 ${name} completed!` : `${name} → Stage ${d.student.currentStage}`, description: done ? "Ready for certificate." : `Stage ${d.student.currentStage}/6.` }); await load();
   278→    } catch (e) { toast({ title: "Failed", description: (e as Error).message, variant: "destructive" }); }
   279→    finally { setBusy(null); }
   280→  }
   281→
   282→  return (
   283→    <div>
   284→      <h3 className="font-bold text-lg mb-1">Student Management</h3>
   285→      <p className="text-xs text-muted-foreground mb-4">Filter, approve payments, advance stages, generate certificates.</p>
   286→      <div className="flex flex-wrap gap-2 mb-3">
   287→        <div className="relative flex-1 min-w-[180px]">
   288→          <Search className="size-4 absolute left-2.5 top-2.5 text-muted-foreground" />
   289→          <Input value={q} onChange={e => setQ(e.target.value)} placeholder="Search name / email / project…" className="bg-muted border-border pl-8 h-9 text-sm" />
   290→        </div>
   291→        <select value={filterStatus} onChange={e => setFilterStatus(e.target.value)} className="bg-muted border border-border rounded-md h-9 px-2 text-xs text-foreground">
   292→          <option value="all">All status</option><option value="pending">Pending</option><option value="active">Active</option><option value="completed">Completed</option>
   293→        </select>
   294→        <select value={filterCollege} onChange={e => setFilterCollege(e.target.value)} className="bg-muted border border-border rounded-md h-9 px-2 text-xs text-foreground max-w-[160px]">
   295→          <option value="all">All colleges</option>{colleges.map(c => <option key={c} value={c}>{c}</option>)}
   296→        </select>
   297→        <Button variant="ghost" size="sm" onClick={load} disabled={loading} className="text-foreground hover:text-cyan-500 dark:text-cyan-400 h-9"><RefreshCw className={cn("size-3.5", loading && "animate-spin")} /></Button>
   298→      </div>
   299→      {loading ? <Loading /> : filtered.length === 0 ? <Empty text="No students match the filters." /> : (
   300→        <div className="border border-border rounded-lg max-h-[60vh] overflow-y-auto">
   301→          <table className="w-full text-sm">
   302→            <thead className="sticky top-0 bg-popover z-10"><tr className="text-left text-[10px] text-muted-foreground border-b border-border">
   303→              <th className="px-3 py-2">Student</th><th className="px-3 py-2 hidden md:table-cell">Project</th><th className="px-3 py-2">Status</th><th className="px-3 py-2 text-right">Action</th>
   304→            </tr></thead>
   305→            <tbody>
   306→              {filtered.map(s => {
   307→                const done = s.internshipStatus === "completed";
   308→                const inProg = s.internshipStatus === "in_progress";
   309→                return (
   310→                  <tr key={s.id} className="border-b border-border/30 last:border-0 hover:bg-secondary/40">
   311→                    <td className="px-3 py-2"><div className="font-medium text-foreground text-xs">{s.firstName} {s.lastName}</div><div className="text-[10px] text-muted-foreground">{s.degree} · {s.college}</div></td>
   312→                    <td className="px-3 py-2 hidden md:table-cell text-xs text-foreground max-w-[180px]"><div className="truncate" title={s.project}>{s.project}</div><div className="text-[10px] text-muted-foreground">{s.categoryLabel}</div></td>
   313→                    <td className="px-3 py-2">{done ? <Badge color="emerald" icon={<CheckCircle2 className="size-2.5" />}>Done</Badge> : inProg ? <Badge color="cyan" icon={<Loader2 className="size-2.5" />}>Stage {s.currentStage}/6</Badge> : <Badge color="amber" icon={<Lock className="size-2.5" />}>Pending</Badge>}</td>
   314→                    <td className="px-3 py-2 text-right whitespace-nowrap">
   315→                      {done ? <GenCertButton studentId={s.id} name={s.firstName} small /> :
   316→                        inProg ? <Button size="sm" variant="outline" onClick={() => advance(s.id, s.firstName)} disabled={busy === s.id + "-a"} className="h-7 text-[11px] bg-secondary border-border text-foreground hover:text-cyan-500 dark:text-cyan-400">{busy === s.id + "-a" ? <Loader2 className="size-3 animate-spin" /> : <Forward className="size-3.5" />}Advance</Button> :
   317→                        <Button size="sm" onClick={() => start(s.id, s.firstName)} disabled={busy === s.id} className="h-7 text-[11px] bg-gradient-to-r from-emerald-600 to-cyan-500 text-primary-foreground hover:opacity-90">{busy === s.id ? <Loader2 className="size-3 animate-spin" /> : <Rocket className="size-3.5" />}Approve & Start</Button>}
   318→                    </td>
   319→                  </tr>
   320→                );
   321→              })}
   322→            </tbody>
   323→          </table>
   324→        </div>
   325→      )}
   326→    </div>
   327→  );
   328→}
